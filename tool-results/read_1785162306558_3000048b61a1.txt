    27→    public static function render_page() {
    28→        self::handle_save();

    31→        $edit_plan  = null;
    32→        $is_editing = false;
    33→        $edit_id    = isset( $_GET['edit'] ) ? absint( $_GET['edit'] ) : 0;
    34→
    35→        if ( $edit_id > 0 ) {
    36→            global $wpdb;
    37→            $table     = $wpdb->prefix . 'bv_plans';
    38→            $edit_plan = $wpdb->get_row( $wpdb->prepare( "SELECT * FROM $table WHERE id = %d", $edit_id ) );
    39→            if ( $edit_plan ) {
    40→                $is_editing = true;
    41→            }
    42→        }
    43→
    44→        if ( $is_editing && $edit_plan ) {
    45→            self::render_form( $edit_plan );
    46→        } else {
    47→            self::render_list();
    48→        }
    49→    }

   457→        }
   458→
   459→        if ( ! isset( $_POST['bv_plan_nonce'] ) || ! wp_verify_nonce( $_POST['bv_plan_nonce'], 'bv_save_plan' ) ) {
   460→            wp_die( 'Security check failed.' );
   461→        }
   462→
   463→        global $wpdb;
   464→        $table          = $wpdb->prefix . 'bv_plans';
   465→        $features_table = $wpdb->prefix . 'bv_plan_features';
   466→
   467→        $name      = sanitize_text_field( wp_unslash( $_POST['name'] ) );
   468→        $subtitle  = sanitize_text_field( wp_unslash( $_POST['subtitle'] ?? '' ) );
   469→        $price     = ! empty( $_POST['price'] ) ? floatval( $_POST['price'] ) : 0;
   470→        $color     = ! empty( $_POST['color'] ) ? sanitize_hex_color( wp_unslash( $_POST['color'] ) ) : '#002B5C';
   471→        $btn_label = sanitize_text_field( wp_unslash( $_POST['button_label'] ?? 'GET STARTED' ) );
   472→        $btn_type  = in_array( $_POST['button_type'] ?? 'cart', array( 'cart', 'quote', 'booking', 'link' ), true )
   473→                     ? $_POST['button_type'] : 'cart';
   474→        $btn_url   = esc_url_raw( wp_unslash( $_POST['button_url'] ?? '' ) );
   475→        $wc_id     = sanitize_text_field( wp_unslash( $_POST['woocommerce_product_id'] ?? '' ) );
   476→        $cat_id    = ! empty( $_POST['category_id'] ) ? absint( $_POST['category_id'] ) : null;
   477→        $visible   = isset( $_POST['visible'] ) ? 1 : 0;
   478→        $featured  = isset( $_POST['featured'] ) ? 1 : 0;
   479→        $order     = absint( $_POST['display_order'] ?? 0 );
   480→
   481→        // Features from the dynamic form.
   482→        $features_input = isset( $_POST['features'] ) && is_array( $_POST['features'] )
   483→            ? array_map( 'sanitize_text_field', wp_unslash( $_POST['features'] ) )
   484→            : array();
   485→
   486→        if ( empty( $name ) ) {
   487→            wp_admin_notice( 'Plan name is required.', array( 'type' => 'error' ) );
   488→            return;
   489→        }
   490→
   491→        $edit_id = isset( $_POST['plan_id'] ) ? absint( $_POST['plan_id'] ) : 0;
   492→
   493→        $data = array(
   494→            'name'                  => $name,
   495→            'subtitle'              => $subtitle,
   496→            'price'                 => $price,
   497→            'color'                 => $color,
   498→            'button_label'          => $btn_label,
   499→            'button_type'           => $btn_type,
   500→            'button_url'            => $btn_url,
   501→            'woocommerce_product_id' => $wc_id,
   502→            'category_id'           => $cat_id,
   503→            'visible'               => $visible,
   504→            'featured'              => $featured,
   505→            'display_order'         => $order,
   506→        );
   507→        $format = array( '%s', '%s', '%f', '%s', '%s', '%s', '%s', '%s', '%d', '%d', '%d', '%d' );
   508→
   509→        if ( $edit_id > 0 ) {
   510→            $wpdb->update( $table, $data, array( 'id' => $edit_id ), $format, array( '%d' ) );
   511→
   512→            // Delete old features and insert new ones.
   513→            $wpdb->delete( $features_table, array( 'plan_id' => $edit_id ), array( '%d' ) );
   514→            foreach ( $features_input as $f_text ) {
   515→                if ( ! empty( trim( $f_text ) ) ) {
   516→                    $wpdb->insert(
   517→                        $features_table,
   518→                        array( 'plan_id' => $edit_id, 'text' => $f_text ),
   519→                        array( '%d', '%s' )
   520→                    );
   521→                }
   522→            }
   523→
   524→            wp_admin_notice( 'Plan updated successfully.', array( 'type' => 'success' ) );
   525→        } else {
   526→            $wpdb->insert( $table, $data, $format );
   527→            $edit_id = $wpdb->insert_id;
   528→
   529→            // Insert features.
   530→            foreach ( $features_input as $f_text ) {
   531→                if ( ! empty( trim( $f_text ) ) ) {
   532→                    $wpdb->insert(
   533→                        $features_table,
   534→                        array( 'plan_id' => $edit_id, 'text' => $f_text ),
   535→                        array( '%d', '%s' )
   536→                    );
   537→                }
   538→            }
   539→
   540→            wp_admin_notice( 'Plan created successfully.', array( 'type' => 'success' ) );
   541→        }
   542→
   543→        // Sync WooCommerce product meta.
   544→        if ( ! empty( $wc_id ) && function_exists( 'wc_get_product' ) ) {
   545→            update_post_meta( $wc_id, '_bv_plan_id', $edit_id );
   546→        }
   547→    }

   549→    /**
   550→     * Handle delete plan and its features.
   551→     *
   552→     * @return void
   553→     */
   554→    private static function handle_delete() {
   555→        if ( ! isset( $_GET['action'] ) || $_GET['action'] !== 'bv_delete_plan' ) {
   556→            return;
   557→        }
   558→
   559→        $id = isset( $_GET['id'] ) ? absint( $_GET['id'] ) : 0;
   560→        if ( $id <= 0 ) {
   561→            return;
   562→        }
   563→
   564→        if ( ! isset( $_GET['_wpnonce'] ) || ! wp_verify_nonce( $_GET['_wpnonce'], 'bv_delete_plan_' . $id ) ) {
   565→            wp_die( 'Security check failed.' );
   566→        }
   567→
   568→        global $wpdb;
   569→        $plans_table    = $wpdb->prefix . 'bv_plans';
   570→        $features_table = $wpdb->prefix . 'bv_plan_features';
   571→
   572→        // Clean up WC meta.
   573→        $wc_id = $wpdb->get_var( $wpdb->prepare( "SELECT woocommerce_product_id FROM $plans_table WHERE id = %d", $id ) );
   574→        if ( ! empty( $wc_id ) ) {
   575→            delete_post_meta( $wc_id, '_bv_plan_id' );
   576→        }
   577→
   578→        $wpdb->delete( $features_table, array( 'plan_id' => $id ), array( '%d' ) );
   579→        $wpdb->delete( $plans_table, array( 'id' => $id ), array( '%d' ) );
   580→
   581→        wp_safe_redirect( admin_url( 'admin.php?page=bv-plans&bv_deleted=1' ) );
   582→        exit;
   583→    }

   589→     */
   590→    public static function handle_visibility_toggle() {
   591→        if ( ! isset( $_GET['action'] ) || $_GET['action'] !== 'bv_toggle_plan_visibility' ) {
   592→            return;
   593→        }
   594→
   595→        $id = isset( $_GET['id'] ) ? absint( $_GET['id'] ) : 0;
   596→        if ( $id <= 0 ) {
   597→            return;
   598→        }
   599→
   600→        if ( ! isset( $_GET['_wpnonce'] ) || ! wp_verify_nonce( $_GET['_wpnonce'], 'bv_toggle_visibility_' . $id ) ) {
   601→            wp_die( 'Security check failed.' );
   602→        }
   603→
   604→        global $wpdb;
   605→        $table = $wpdb->prefix . 'bv_plans';
   606→
   607→        $wpdb->query(
   608→            $wpdb->prepare( "UPDATE $table SET visible = IF(visible = 1, 0, 1) WHERE id = %d", $id )
   609→        );
   610→
   611→        wp_safe_redirect( admin_url( 'admin.php?page=bv-plans' ) );
   612→        exit;
   613→    }
