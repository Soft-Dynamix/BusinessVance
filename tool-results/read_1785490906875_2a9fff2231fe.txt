   102→    /**
   103→     * Sanitize all settings on save
   104→     */
   105→    public function sanitize_settings( $input ) {
   106→        $sanitized = array();
   107→        $defaults  = self::get_defaults();
   108→
   109→        foreach ( $defaults as $key => $default ) {
   110→            if ( isset( $input[ $key ] ) ) {
   111→                $sanitized[ $key ] = sanitize_text_field( $input[ $key ] );
   112→            } else {
   113→                $sanitized[ $key ] = $default;
   114→            }
   115→        }
   116→
   117→        // Allow longer text for agreement and email bodies
   118→        if ( isset( $input['agreement_text'] ) ) {
   119→            $sanitized['agreement_text'] = wp_kses_post( $input['agreement_text'] );
   120→        }
   121→        if ( isset( $input['portal_welcome_message'] ) ) {
   122→            $sanitized['portal_welcome_message'] = sanitize_textarea_field( $input['portal_welcome_message'] );
   123→        }
   124→        if ( isset( $input['email_project_created_body'] ) ) {
   125→            $sanitized['email_project_created_body'] = sanitize_textarea_field( $input['email_project_created_body'] );
   126→        }
   127→        if ( isset( $input['email_agreement_ready_body'] ) ) {
   128→            $sanitized['email_agreement_ready_body'] = sanitize_textarea_field( $input['email_agreement_ready_body'] );
   129→        }
   130→        if ( isset( $input['email_report_ready_body'] ) ) {
   131→            $sanitized['email_report_ready_body'] = sanitize_textarea_field( $input['email_report_ready_body'] );
   132→        }
   133→        if ( isset( $input['physical_address'] ) ) {
   134→            $sanitized['physical_address'] = sanitize_textarea_field( $input['physical_address'] );
   135→        }
   136→
   137→        // Sanitize URLs
   138→        if ( isset( $input['logo_url'] ) ) {
   139→            $sanitized['logo_url'] = esc_url_raw( $input['logo_url'] );
   140→        }
   141→        if ( isset( $input['favicon_url'] ) ) {
   142→            $sanitized['favicon_url'] = esc_url_raw( $input['favicon_url'] );
   143→        }
   144→
   145→        return $sanitized;
   146→    }

   176→    /**
   177→     * Get all settings (merged with defaults)
   178→     *
   179→     * @return array
   180→     */
   181→    public static function get_settings() {
   182→        $saved   = get_option( self::OPTION_KEY, array() );
   183→        $defaults = self::get_defaults();
   184→
   185→        return array_merge( $defaults, $saved );
   186→    }
