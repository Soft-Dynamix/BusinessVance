   152→    /**
   153→     * Verify admin nonce
   154→     */
   155→    private function verify_nonce() {
   156→        if ( ! check_ajax_referer( 'bv_admin_nonce', 'nonce', false ) ) {
   157→            wp_send_json_error( array( 'message' => 'Security check failed.' ), 403 );
   158→        }
   159→        if ( ! current_user_can( 'manage_options' ) ) {
   160→            wp_send_json_error( array( 'message' => 'Permission denied.' ), 403 );
   161→        }
   162→    }

   776→    /**
   777→     * Get service data for editing
   778→     */
   779→    public function ajax_get_service() {
   780→        $this->verify_nonce();
   781→        global $wpdb;
   782→        $tables = $this->get_tables();
   783→
   784→        $id = intval( $_POST['id'] ?? 0 );
   785→        if ( $id <= 0 ) {
   786→            wp_send_json_error( array( 'message' => 'Invalid ID.' ) );
   787→        }
   788→
   789→        $service = $wpdb->get_row( $wpdb->prepare(
   790→            "SELECT * FROM {$tables['services']} WHERE id = %d",
   791→            $id
   792→        ), ARRAY_A );
   793→
   794→        if ( ! $service ) {
   795→            wp_send_json_error( array( 'message' => 'Service not found.' ) );
   796→        }
   797→
   798→        wp_send_json_success( $service );
   799→    }

   801→    /**
   802→     * Save (create/update) service
   803→     */
   804→    public function ajax_save_service() {
   805→        $this->verify_nonce();
   806→        global $wpdb;
   807→        $tables = $this->get_tables();
   808→
   809→        $id = intval( $_POST['id'] ?? 0 );
   810→        $data = array(
   811→            'name'          => sanitize_text_field( $_POST['name'] ?? '' ),
   812→            'description'   => sanitize_textarea_field( $_POST['description'] ?? '' ),
   813→            'price'         => sanitize_text_field( $_POST['price'] ?? 'R0' ),
   814→            'price_display' => sanitize_text_field( $_POST['price_display'] ?? '' ),
   815→            'icon'          => sanitize_text_field( $_POST['icon'] ?? 'briefcase' ),
   816→            'button_label'  => sanitize_text_field( $_POST['button_label'] ?? 'Get Started' ),
   817→            'service_type'  => sanitize_text_field( $_POST['service_type'] ?? 'onceoff' ),
   818→            'woo_product_id' => intval( $_POST['woo_product_id'] ?? 0 ),
   819→            'category_id'   => intval( $_POST['category_id'] ?? 0 ),
   820→            'is_visible'    => intval( $_POST['is_visible'] ?? 0 ),
   821→            'is_featured'   => intval( $_POST['is_featured'] ?? 0 ),
   822→        );
   823→        $format = array( '%s','%s','%s','%s','%s','%s','%s','%d','%d','%d','%d' );
   824→
   825→        if ( empty( $data['name'] ) ) {
   826→            wp_send_json_error( array( 'message' => __( 'Service name is required.', 'businessvance-services-manager' ) ) );
   827→        }
   828→
   829→        if ( $id > 0 ) {
   830→            // Update

   947→    public function ajax_save_plan() {
   948→        $this->verify_nonce();
   949→        global $wpdb;
   950→        $tables = $this->get_tables();
   951→
   952→        $id = intval( $_POST['id'] ?? 0 );
   953→        $data = array(
   954→            'name'          => sanitize_text_field( $_POST['name'] ?? '' ),
   955→            'subtitle'      => sanitize_text_field( $_POST['subtitle'] ?? '' ),
   956→            'price'         => sanitize_text_field( $_POST['price'] ?? 'R0/mo' ),
   957→            'color'         => sanitize_text_field( $_POST['color'] ?? '#008080' ),
   958→            'button_label'  => sanitize_text_field( $_POST['button_label'] ?? 'Subscribe Now' ),
   959→            'woo_product_id' => intval( $_POST['woo_product_id'] ?? 0 ),
   960→            'category_id'   => intval( $_POST['category_id'] ?? 0 ),
   961→            'is_visible'    => intval( $_POST['is_visible'] ?? 0 ),
   962→            'is_featured'   => intval( $_POST['is_featured'] ?? 0 ),
   963→        );
   964→        $format = array( '%s','%s','%s','%s','%s','%d','%d','%d','%d' );
   965→
   966→        $features = $_POST['features'] ?? array();
   967→        if ( ! is_array( $features ) ) {
   968→            $features = array();
   969→        }
   970→        $features = array_map( 'sanitize_text_field', $features );
   971→
   972→        if ( empty( $data['name'] ) ) {
   973→            wp_send_json_error( array( 'message' => __( 'Plan name is required.', 'businessvance-services-manager' ) ) );
   974→        }
   975→
   976→        if ( $id > 0 ) {
   977→            // Update plan
   978→            $wpdb->update( $tables['plans'], $data, array( 'id' => $id ), $format, array( '%d' ) );
   979→
   980→            // Update features - delete existing and re-insert
   981→            $wpdb->delete( $tables['features'], array( 'plan_id' => $id ), array( '%d' ) );
   982→            foreach ( $features as $i => $feature_text ) {
   983→                if ( ! empty( $feature_text ) ) {
   984→                    $wpdb->insert( $tables['features'], array(
   985→                        'plan_id'       => $id,
   986→                        'feature_text'  => $feature_text,
   987→                        'display_order' => $i,
   988→                    ), array( '%d', '%s', '%d' ) );
   989→                }
   990→            }
   991→
   992→            wp_send_json_success( array( 'message' => __( 'Plan updated.', 'businessvance-services-manager' ), 'id' => $id ) );
   993→        } else {
   994→            // Get max display_order
   995→            $max_order = (int) $wpdb->get_var( "SELECT COALESCE(MAX(display_order), 0) FROM {$tables['plans']}" );
   996→            $data['display_order'] = $max_order + 1;
   997→            $format[] = '%d';
   998→
   999→            $wpdb->insert( $tables['plans'], $data, $format );
  1000→            $new_id = $wpdb->insert_id;
  1001→
  1002→            // Insert features
  1003→            foreach ( $features as $i => $feature_text ) {
  1004→                if ( ! empty( $feature_text ) ) {
  1005→                    $wpdb->insert( $tables['features'], array(
  1006→                        'plan_id'       => $new_id,
