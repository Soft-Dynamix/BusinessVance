    44→    private function verify_project_access( $project_id ) {
    45→        if ( ! is_user_logged_in() ) return false;
    46→        global $wpdb;
    47→        $user_id = get_current_user_id();
    48→        $project = $wpdb->get_row( $wpdb->prepare(
    49→            "SELECT * FROM {$wpdb->prefix}bv_projects WHERE id = %d AND client_user_id = %d",
    50→            $project_id, $user_id
    51→        ) );
    52→        return $project ? $project : false;

    55→    public function render_portal( $atts ) {
    56→        if ( ! is_user_logged_in() ) {
    57→            return '<div class="bv-portal-login-message"><p>Please <a href="' . esc_url( wp_login_url( get_permalink() ) ) . '">log in</a> to access your client portal.</p></div>';
    58→        }
    59→
    60→        global $wpdb;
    61→        $user_id    = get_current_user_id();
    62→        $user       = wp_get_current_user();
    63→        $project_id = isset( $_GET['project_id'] ) ? absint( $_GET['project_id'] ) : 0;
    64→        $tab        = isset( $_GET['tab'] ) ? sanitize_text_field( $_GET['tab'] ) : 'overview';
    65→
    66→        // Get all projects for this user
    67→        $projects_table = $wpdb->prefix . 'bv_projects';
    68→        $projects = $wpdb->get_results( $wpdb->prepare(
    69→            "SELECT * FROM {$projects_table} WHERE client_user_id = %d ORDER BY created_at DESC",
    70→            $user_id
    71→        ) );
    72→
    73→        // No projects yet
    74→        if ( empty( $projects ) ) {
    75→            return $this->render_no_projects( $user );
    76→        }
    77→
    78→        // Validate project_id belongs to this user
    79→        $active_project = null;
    80→        if ( $project_id ) {
    81→            foreach ( $projects as $p ) {
    82→                if ( $p->id == $project_id ) { $active_project = $p; break; }
    83→            }
    84→        }
    85→        if ( ! $active_project ) {
    86→            $active_project = $projects[0];
    87→            $project_id = $active_project->id;
    88→        }
    89→
    90→        // Load project data
    91→        $services   = $wpdb->get_results( $wpdb->prepare(
    92→            "SELECT s.* FROM {$wpdb->prefix}bv_project_services ps JOIN {$wpdb->prefix}bv_services s ON ps.service_id = s.id WHERE ps.project_id = %d", $project_id ) );
    93→        $agreement  = $wpdb->get_row( $wpdb->prepare(
    94→            "SELECT * FROM {$wpdb->prefix}bv_project_agreements WHERE project_id = %d ORDER BY id DESC LIMIT 1", $project_id ) );
    95→        $documents  = $wpdb->get_results( $wpdb->prepare(
    96→            "SELECT * FROM {$wpdb->prefix}bv_project_documents WHERE project_id = %d ORDER BY created_at DESC", $project_id ) );
    97→        $reports    = $wpdb->get_results( $wpdb->prepare(
    98→            "SELECT * FROM {$wpdb->prefix}bv_project_reports WHERE project_id = %d ORDER BY created_at DESC", $project_id ) );
    99→        $messages   = $wpdb->get_results( $wpdb->prepare(
   100→            "SELECT * FROM {$wpdb->prefix}bv_project_messages WHERE project_id = %d ORDER BY created_at ASC", $project_id ) );
   101→
   102→        // Mark messages as read
   103→        $wpdb->update( $wpdb->prefix . 'bv_project_messages',
   104→            array( 'is_read' => 1 ),
   105→            array( 'project_id' => $project_id, 'sender_type' => 'admin' ),
   106→            array( '%d', '%d' )
   107→        );
   108→
   109→        // Get questionnaire data
   110→        $questionnaire = $this->get_questionnaire_data( $project_id );
   111→
   112→        ob_start();
   113→        ?>
   114→        <div class="bv-portal" id="bv-portal-app">
   115→            <style><?php echo $this->get_inline_css(); ?></style>
   116→
   117→            <!-- Header -->
   118→            <div class="bv-portal-header">
   119→                <div class="bv-portal-header-inner">
   120→                    <div class="bv-portal-brand">
   121→                        <span class="bv-portal-logo">✦</span>
   122→                        <h1>BusinessVance</h1>
   123→                    </div>
   124→                    <div class="bv-portal-user-info">
   125→                        <span>Welcome, <strong><?php echo esc_html( $user->display_name ); ?></strong></span>
   126→                        <?php if ( $active_project->client_company ) : ?>
   127→                            <span class="bv-portal-company"><?php echo esc_html( $active_project->client_company ); ?></span>
   128→                        <?php endif; ?>
   129→                    </div>
   130→                </div>
   131→            </div>
   132→
   133→            <div class="bv-portal-body">
   134→                <!-- Sidebar: Project List -->
   135→                <div class="bv-portal-sidebar">
   136→                    <h3>My Projects</h3>
   137→                    <div class="bv-portal-project-list">
   138→                        <?php foreach ( $projects as $p ) : ?>
   139→                        <a href="?project_id=<?php echo $p->id; ?>" class="bv-portal-project-item <?php echo $p->id == $project_id ? 'active' : ''; ?>">
   140→                            <span class="bv-portal-project-num"><?php echo esc_html( $p->project_number ); ?></span>
   141→                            <span class="bv-portal-project-status-badge bv-status-<?php echo esc_attr( $p->status ); ?>">
   142→                                <?php echo esc_html( $this->status_label( $p->status ) ); ?>
   143→                            </span>
   144→                        </a>
   145→                        <?php endforeach; ?>
   146→                    </div>
   147→                </div>
   148→
   149→                <!-- Main Content -->
   150→                <div class="bv-portal-main">
   151→                    <!-- Tabs -->
   152→                    <div class="bv-portal-tabs">
   153→                        <?php
   154→                        $tabs = array( 'overview' => 'Overview', 'agreement' => 'Agreement', 'questionnaire' => 'Questionnaire', 'documents' => 'Documents', 'reports' => 'Reports', 'messages' => 'Messages' );
   155→                        foreach ( $tabs as $tab_id => $tab_label ) :
   156→                            $active = ( $tab === $tab_id ) ? 'active' : '';
   157→                            $count = '';
   158→                            if ( $tab_id === 'messages' ) {
   159→                                $unread = $wpdb->get_var( $wpdb->prepare(
   160→                                    "SELECT COUNT(*) FROM {$wpdb->prefix}bv_project_messages WHERE project_id = %d AND sender_type = 'admin' AND is_read = 0",
   161→                                    $project_id ) );
   162→                                if ( $unread ) $count = " <span class=\"bv-badge\">{$unread}</span>";
   163→                            }
   164→                            if ( $tab_id === 'reports' && ! empty( $reports ) ) {
   165→                                $count = ' <span class="bv-count">' . count( $reports ) . '</span>';
   166→                            }
   167→                            if ( $tab_id === 'documents' && ! empty( $documents ) ) {
   168→                                $count = ' <span class="bv-count">' . count( $documents ) . '</span>';
   169→                            }
   170→                        ?>
   171→                        <a href="?project_id=<?php echo $project_id; ?>&tab=<?php echo $tab_id; ?>" class="bv-portal-tab <?php echo $active; ?>"><?php echo $tab_label . $count; ?></a>
   172→                        <?php endforeach; ?>
   173→                    </div>
   174→
   175→                    <div class="bv-portal-content">
   176→                        <?php
   177→                        switch ( $tab ) {
   178→                            case 'overview': echo $this->render_overview_tab( $active_project, $services ); break;
   179→                            case 'agreement': echo $this->render_agreement_tab( $active_project, $agreement ); break;
   180→                            case 'questionnaire': echo $this->render_questionnaire_tab( $active_project, $questionnaire ); break;
   181→                            case 'documents': echo $this->render_documents_tab( $active_project, $documents ); break;
   182→                            case 'reports': echo $this->render_reports_tab( $active_project, $reports ); break;
   183→                            case 'messages': echo $this->render_messages_tab( $active_project, $messages ); break;
   184→                            default: echo $this->render_overview_tab( $active_project, $services );
   185→                        }
   186→                        ?>
   187→                    </div>
   188→                </div>
   189→            </div>
   190→
   191→            <div class="bv-portal-footer">
   192→                <p>&copy; <?php echo date('Y'); ?> BusinessVance Consulting. All rights reserved.</p>
   193→            </div>
   194→        </div>
   195→        <?php
   196→        return ob_get_clean();
   197→    }

   331→    private function get_questionnaire_data( $project_id ) {
   332→        global $wpdb;
   333→        $sections = $wpdb->get_results(
   334→            "SELECT qs.* FROM {$wpdb->prefix}bv_questionnaire_sections qs
   335→             JOIN {$wpdb->prefix}bv_questionnaire_templates qt ON qs.template_id = qt.id
   336→             WHERE qt.status = 'published'
   337→             ORDER BY qs.display_order ASC"
   338→        );
   339→        foreach ( $sections as &$section ) {
   340→            $section->questions = $wpdb->get_results( $wpdb->prepare(
   341→                "SELECT q.*, r.response_value
   342→                 FROM {$wpdb->prefix}bv_questionnaire_questions q
   343→                 LEFT JOIN {$wpdb->prefix}bv_questionnaire_responses r
   344→                   ON r.question_id = q.id AND r.project_id = %d
   345→                 WHERE q.section_id = %d
   346→                 ORDER BY q.display_order ASC",
   347→                $project_id, $section->id
   348→            ) );
   349→        }
   350→        return $sections;
   351→    }

   570→    public function ajax_upload_document() {
   571→        check_ajax_referer( 'bv_portal_action', 'nonce' );
   572→        if ( ! is_user_logged_in() ) wp_send_json_error( 'Not logged in' );
   573→
   574→        $project_id = absint( $_POST['project_id'] );
   575→        $project = $this->verify_project_access( $project_id );
   576→        if ( ! $project ) wp_send_json_error( 'Project not found or access denied' );
   577→
   578→        if ( empty( $_FILES['file'] ) ) wp_send_json_error( 'No file uploaded' );
   579→
   580→        $file = $_FILES['file'];
   581→        $allowed = array( 'pdf', 'doc', 'docx', 'jpg', 'jpeg', 'png' );
   582→        $ext = strtolower( pathinfo( $file['name'], PATHINFO_EXTENSION ) );
   583→        if ( ! in_array( $ext, $allowed ) ) wp_send_json_error( 'File type not allowed' );
   584→
   585→        $filename = $project_id . '_' . time() . '_' . sanitize_file_name( $file['name'] );
   586→        $upload_path = BV_UPLOAD_DIR . '/' . $filename;
   587→
   588→        if ( ! move_uploaded_file( $file['tmp_name'], $upload_path ) ) {
   589→            wp_send_json_error( 'Upload failed' );
   590→        }
   591→
   592→        global $wpdb;
   593→        $wpdb->insert( $wpdb->prefix . 'bv_project_documents', array(
   594→            'project_id'  => $project_id,
   595→            'service_id'  => 0,
   596→            'name'        => sanitize_text_field( $_POST['name'] ?? $file['name'] ),
   597→            'filename'    => $filename,
   598→            'filepath'    => $upload_path,
   599→            'filesize'    => $file['size'],
   600→            'mime_type'   => $file['type'],
   601→            'category'    => sanitize_text_field( $_POST['category'] ?? 'other' ),
   602→            'uploaded_by' => 'client',
   603→        ), array( '%d', '%d', '%s', '%s', '%s', '%d', '%s', '%s', '%s' ) );
   604→
   605→        $wpdb->insert( $wpdb->prefix . 'bv_activity_log', array(
   606→            'project_id'  => $project_id,
   607→            'entity_type' => 'document',
   608→            'entity_id'   => $wpdb->insert_id,
   609→            'action'      => 'uploaded',
   610→            'description' => 'Client uploaded document: ' . sanitize_text_field( $_POST['name'] ?? $file['name'] ),
   611→            'user_id'     => get_current_user_id(),
   612→        ), array( '%d', '%s', '%d', '%s', '%s', '%d' ) );
   613→
   614→        wp_send_json_success( 'Document uploaded successfully' );
   615→    }

   617→    public function ajax_submit_questionnaire() {
   618→        check_ajax_referer( 'bv_portal_action', 'nonce' );
   619→        if ( ! is_user_logged_in() ) wp_send_json_error( 'Not logged in' );
   620→
   621→        $project_id = absint( $_POST['project_id'] );
   622→        $project = $this->verify_project_access( $project_id );
   623→        if ( ! $project ) wp_send_json_error( 'Project not found or access denied' );
   624→
   625→        global $wpdb;
   626→        $responses_table = $wpdb->prefix . 'bv_questionnaire_responses';
   627→
   628→        foreach ( $_POST['responses'] as $question_id => $value ) {
   629→            if ( is_array( $value ) ) $value = wp_json_encode( $value );
   630→            $q_id = absint( $question_id );
   631→            $existing = $wpdb->get_var( $wpdb->prepare(
   632→                "SELECT id FROM {$responses_table} WHERE project_id = %d AND question_id = %d",
   633→                $project_id, $q_id
   634→            ) );
   635→            $data = array(
   636→                'project_id'     => $project_id,
   637→                'service_id'     => 0,
   638→                'question_id'    => $q_id,
   639→                'response_value' => sanitize_text_field( $value ),
   640→            );
   641→            $format = array( '%d', '%d', '%d', '%s' );
   642→            if ( $existing ) {
   643→                $wpdb->update( $responses_table, $data, array( 'id' => $existing ), $format, array( '%d' ) );
   644→            } else {
   645→                $wpdb->insert( $responses_table, $data, $format );
   646→            }
   647→        }
   648→
   649→        // Update project status if still awaiting questionnaire
   650→        if ( $project->status === 'awaiting-questionnaire' ) {
   651→            $wpdb->update( $wpdb->prefix . 'bv_projects',
   652→                array( 'status' => 'awaiting-documents', 'progress_percent' => 25 ),
   653→                array( 'id' => $project_id ),
   654→                array( '%s', '%d' ), array( '%d' )
   655→            );
   656→        }
   657→
   658→        $wpdb->insert( $wpdb->prefix . 'bv_activity_log', array(
   659→            'project_id' => $project_id, 'entity_type' => 'questionnaire', 'entity_id' => $project_id,
   660→            'action' => 'submitted', 'description' => 'Client submitted questionnaire responses', 'user_id' => get_current_user_id(),
   661→        ), array( '%d', '%s', '%d', '%s', '%s', '%d' ) );
   662→
   663→        wp_send_json_success( 'Questionnaire saved successfully' );
   664→    }

   735→    public function ajax_download_report() {
   736→        check_ajax_referer( 'bv_portal_action', 'nonce' );
   737→        if ( ! is_user_logged_in() ) die( 'Not logged in' );
   738→
   739→        $report_id = absint( $_GET['report_id'] ?? $_POST['report_id'] ?? 0 );
   740→        if ( ! $report_id ) die( 'Invalid report' );
   741→
   742→        global $wpdb;
   743→        $report = $wpdb->get_row( $wpdb->prepare(
   744→            "SELECT r.* FROM {$wpdb->prefix}bv_project_reports r
   745→             JOIN {$wpdb->prefix}bv_projects p ON r.project_id = p.id
   746→             WHERE r.id = %d AND p.client_user_id = %d AND r.status = 'delivered'",
   747→            $report_id, get_current_user_id()
   748→        ) );
   749→        if ( ! $report || ! file_exists( $report->filepath ) ) {
   750→            wp_die( 'Report not found' );
   751→        }
   752→
   753→        header( 'Content-Type: ' . $report->mime_type );
   754→        header( 'Content-Disposition: attachment; filename="' . basename( $report->filename ) . '"' );
   755→        header( 'Content-Length: ' . $report->filesize );
   756→        readfile( $report->filepath );
   757→        exit;
   758→    }
