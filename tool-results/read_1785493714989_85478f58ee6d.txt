   138→function StatusBadge({ status }: { status: string }) {
   139→  const config = statusConfig[status] || statusConfig.draft
   140→  return (
   141→    <Badge variant="outline" className={`${config.className} gap-1 font-medium`}>
   142→      {config.icon}
   143→      {config.label}
   144→    </Badge>
   145→  )
   146→}

   148→function QuestionTypeBadge({ type }: { type: string }) {
   149→  const colors: Record<string, string> = {
   150→    text: 'bg-sky-50 text-sky-700 border-sky-200',
   151→    textarea: 'bg-sky-50 text-sky-700 border-sky-200',
   152→    number: 'bg-violet-50 text-violet-700 border-violet-200',
   153→    email: 'bg-cyan-50 text-cyan-700 border-cyan-200',
   154→    phone: 'bg-cyan-50 text-cyan-700 border-cyan-200',
   155→    date: 'bg-orange-50 text-orange-700 border-orange-200',
   156→    select: 'bg-pink-50 text-pink-700 border-pink-200',
   157→    multiselect: 'bg-pink-50 text-pink-700 border-pink-200',
   158→    radio: 'bg-rose-50 text-rose-700 border-rose-200',
   159→    checkbox: 'bg-rose-50 text-rose-700 border-rose-200',
   160→    file: 'bg-amber-50 text-amber-700 border-amber-200',
   161→    heading: 'bg-gray-100 text-gray-500 border-gray-200',
   162→    paragraph: 'bg-gray-100 text-gray-500 border-gray-200',
   163→  }
   164→  return (
   165→    <Badge variant="outline" className={`${colors[type] || 'bg-gray-100 text-gray-600'} text-xs font-medium`}>
   166→      {type}
   167→    </Badge>
   168→  )
   169→}

   171→function parseOptions(optionsStr: string): string[] {
   172→  try {
   173→    const parsed = JSON.parse(optionsStr)
   174→    return Array.isArray(parsed) ? parsed : []
   175→  } catch {
   176→    return []
   177→  }
   178→}

   183→  const [activeTab, setActiveTab] = useState('questionnaires')
   184→

   186→  const [questionnaireTemplates, setQuestionnaireTemplates] = useState<QuestionnaireTemplate[]>([])

   189→

   191→  const [viewQuestionnaire, setViewQuestionnaire] = useState<QuestionnaireTemplate | null>(null)

   197→

   203→  // Loading states

   205→

   207→    try {
   208→      const [qRes, aRes] = await Promise.all([
   209→        fetch('/api/questionnaire-templates'),
   210→        fetch('/api/agreement-templates'),
   211→      ])
   212→      if (qRes.ok) {
   213→        const qData = await qRes.json()

   218→        setAgreementTemplates(aData.templates || [])

   220→    } catch (e) {
   221→      console.error(e)

   237→        const data = await res.json()

   241→      console.error(e)
   242→    }
   243→  }
   244→
   245→  const handleDelete = async () => {
   246→    if (!deleteTarget) return

   248→    try {
   249→      const endpoint = deleteTarget.type === 'questionnaire'
   250→        ? `/api/questionnaire-templates/${deleteTarget.id}`
   251→        : `/api/agreement-templates/${deleteTarget.id}`
   252→      const res = await fetch(endpoint, { method: 'DELETE' })
   253→      if (res.ok) {
   254→        setDeleteTarget(null)
   255→        fetchData()
   256→      }
   257→    } catch (e) {
   258→      console.error(e)
   259→    } finally {
   260→      setActionLoading(false)
   261→    }
   262→  }
   263→
   264→  const handleOpenEditAgreement = (template: AgreementTemplate) => {

   266→    setEditAgreement(template)
   267→  }
   268→
   269→  const handleSaveEditAgreement = async () => {
   270→    if (!editAgreement) return
   271→    setActionLoading(true)
   272→    try {
   273→      const res = await fetch(`/api/agreement-templates/${editAgreement.id}`, {
   274→        method: 'PUT',
   275→        headers: { 'Content-Type': 'application/json' },
   276→        body: JSON.stringify(editForm),
   277→      })
   278→      if (res.ok) {
   279→        setEditAgreement(null)
   280→        fetchData()
   281→      }
   282→    } catch (e) {
   283→      console.error(e)
   284→    } finally {
   285→      setActionLoading(false)
   286→    }
   287→  }
   288→
   289→  const handleAddQuestionnaire = async () => {
   290→    if (!addQForm.name.trim()) return
   291→    setActionLoading(true)
   292→    try {
   293→      const res = await fetch('/api/questionnaire-templates', {
   294→        method: 'POST',
   295→        headers: { 'Content-Type': 'application/json' },
   296→        body: JSON.stringify(addQForm),
   297→      })
   298→      if (res.ok) {
   299→        setAddQuestionnaire(false)
   300→        setAddQForm({ name: '', description: '', version: '1.0', status: 'draft' })
   301→        fetchData()
   302→      }
   303→    } catch (e) {
   304→      console.error(e)
   305→    } finally {
   306→      setActionLoading(false)
   307→    }
   308→  }

   310→  const handleAddAgreement = async () => {

   313→    try {
   314→      const res = await fetch('/api/agreement-templates', {
   315→        method: 'POST',
   316→        headers: { 'Content-Type': 'application/json' },
   317→        body: JSON.stringify(addAForm),
   318→      })

   320→        setAddAgreement(false)
   321→        setAddAForm({ name: '', content: '', version: '1.0', status: 'draft' })

   325→      console.error(e)
