   449→   449→    /**
   450→   450→     * Handle save (create/update) plan and its features.
   451→   451→     *
   452→   452→     * @return void
   453→   453→     */
   454→   454→    private static function handle_save() {
   455→   455→        if ( ! isset( $_POST['bv_action'] ) || $_POST['bv_action'] !== 'bv_save_plan' ) {
   456→   456→            return;
   457→   457→        }
   458→   458→
   459→   459→        if ( ! isset( $_POST['bv_plan_nonce'] ) || ! wp_verify_nonce( $_POST['bv_plan_nonce'], 'bv_save_plan' ) ) {
   460→   460→            wp_die( 'Security check failed.' );
   461→   461→        }
   462→   462→
   463→   463→        global $wpdb;
   464→   464→        $table          = $wpdb->prefix . 'bv_plans';
   465→   465→        $features_table = $wpdb->prefix . 'bv_plan_features';
   466→   466→
   467→   467→        $name      = sanitize_text_field( wp_unslash( $_POST['name'] ) );
   468→   468→        $subtitle  = sanitize_text_field( wp_unslash( $_POST['subtitle'] ?? '' ) );
   469→   469→        $price     = ! empty( $_POST['price'] ) ? floatval( $_POST['price'] ) : 0;
   470→   470→        $color     = ! empty( $_POST['color'] ) ? sanitize_hex_color( wp_unslash( $_POST['color'] ) ) : '#002B5C';
   471→   471→        $btn_label = sanitize_text_field( wp_unslash( $_POST['button_label'] ?? 'GET STARTED' ) );
   472→   472→        $btn_type  = in_array( $_POST['button_type'] ?? 'cart', array( 'cart', 'quote', 'booking', 'link' ), true )
   473→   473→                     ? $_POST['button_type'] : 'cart';
   474→   474→        $btn_url   = esc_url_raw( wp_unslash( $_POST['button_url'] ?? '' ) );
   475→   475→        $wc_id     = sanitize_text_field( wp_unslash( $_POST['woocommerce_product_id'] ?? '' ) );
   476→   476→        $cat_id    = ! empty( $_POST['category_id'] ) ? absint( $_POST['category_id'] ) : null;
   477→   477→        $visible   = isset( $_POST['visible'] ) ? 1 : 0;
   478→   478→        $featured  = isset( $_POST['featured'] ) ? 1 : 0;
   479→   479→        $order     = absint( $_POST['display_order'] ?? 0 );
   480→   480→
   481→   481→        // Features from the dynamic form.
   482→   482→        $features_input = isset( $_POST['features'] ) && is_array( $_POST['features'] )
   483→   483→            ? array_map( 'sanitize_text_field', wp_unslash( $_POST['features'] ) )
   484→   484→            : array();
   485→   485→
   486→   486→        if ( empty( $name ) ) {
   487→   487→            wp_admin_notice( 'Plan name is required.', array( 'type' => 'error' ) );
   488→   488→            return;
   489→   489→        }
   490→   490→
   491→   491→        $edit_id = isset( $_POST['plan_id'] ) ? absint( $_POST['plan_id'] ) : 0;
   492→   492→
   493→   493→        $data = array(
   494→   494→            'name'                  => $name,
   495→   495→            'subtitle'              => $subtitle,
   496→   496→            'price'                 => $price,
   497→   497→            'color'                 => $color,
   498→   498→            'button_label'          => $btn_label,
   499→   499→            'button_type'           => $btn_type,
   500→   500→            'button_url'            => $btn_url,
   501→   501→            'woocommerce_product_id' => $wc_id,
   502→   502→            'category_id'           => $cat_id,
   503→   503→            'visible'               => $visible,
   504→   504→            'featured'              => $featured,
   505→   505→            'display_order'         => $order,
   506→   506→        );
   507→   507→        $format = array( '%s', '%s', '%f', '%s', '%s', '%s', '%s', '%s', '%d', '%d', '%d', '%d' );
   508→   508→
   509→   509→        if ( $edit_id > 0 ) {
   510→   510→            $wpdb->update( $table, $data, array( 'id' => $edit_id ), $format, array( '%d' ) );
   511→   511→
   512→   512→            // Delete old features and insert new ones.
   513→   513→            $wpdb->delete( $features_table, array( 'plan_id' => $edit_id ), array( '%d' ) );
   514→   514→            foreach ( $features_input as $f_text ) {
   515→   515→                if ( ! empty( trim( $f_text ) ) ) {
   516→   516→                    $wpdb->insert(
   517→   517→                        $features_table,
   518→   518→                        array( 'plan_id' => $edit_id, 'text' => $f_text ),
   519→   519→                        array( '%d', '%s' )
   520→   520→                    );
   521→   521→                }
   522→   522→            }
   523→   523→
   524→   524→            wp_admin_notice( 'Plan updated successfully.', array( 'type' => 'success' ) );
   525→   525→        } else {
   526→   526→            $wpdb->insert( $table, $data, $format );
   527→   527→            $edit_id = $wpdb->insert_id;
   528→   528→
   529→   529→            // Insert features.
   530→   530→            foreach ( $features_input as $f_text ) {
   531→   531→                if ( ! empty( trim( $f_text ) ) ) {
   532→   532→                    $wpdb->insert(
   533→   533→                        $features_table,
   534→   534→                        array( 'plan_id' => $edit_id, 'text' => $f_text ),
   535→   535→                        array( '%d', '%s' )
   536→   536→                    );
   537→   537→                }
   538→   538→            }
   539→   539→
   540→   540→            wp_admin_notice( 'Plan created successfully.', array( 'type' => 'success' ) );
   541→   541→        }
   542→   542→
   543→   543→        // Sync WooCommerce product meta.
   544→   544→        if ( ! empty( $wc_id ) && function_exists( 'wc_get_product' ) ) {
   545→   545→            update_post_meta( $wc_id, '_bv_plan_id', $edit_id );
   546→   546→        }
   547→   547→    }

   550→   550→     * Handle delete plan and its features.
   551→   551→     *
   552→   552→     * @return void
   553→   553→     */
   554→   554→    private static function handle_delete() {
   555→   555→        if ( ! isset( $_GET['action'] ) || $_GET['action'] !== 'bv_delete_plan' ) {
   556→   556→            return;
   557→   557→        }
   558→   558→
   559→   559→        $id = isset( $_GET['id'] ) ? absint( $_GET['id'] ) : 0;
   560→   560→        if ( $id <= 0 ) {
   561→   561→            return;
   562→   562→        }
   563→   563→
   564→   564→        if ( ! isset( $_GET['_wpnonce'] ) || ! wp_verify_nonce( $_GET['_wpnonce'], 'bv_delete_plan_' . $id ) ) {
   565→   565→            wp_die( 'Security check failed.' );
   566→   566→        }
   567→   567→
   568→   568→        global $wpdb;
   569→   569→        $plans_table    = $wpdb->prefix . 'bv_plans';
   570→   570→        $features_table = $wpdb->prefix . 'bv_plan_features';
   571→   571→
   572→   572→        // Clean up WC meta.
   573→   573→        $wc_id = $wpdb->get_var( $wpdb->prepare( "SELECT woocommerce_product_id FROM $plans_table WHERE id = %d", $id ) );
   574→   574→        if ( ! empty( $wc_id ) ) {
   575→   575→            delete_post_meta( $wc_id, '_bv_plan_id' );
   576→   576→        }
   577→   577→
   578→   578→        $wpdb->delete( $features_table, array( 'plan_id' => $id ), array( '%d' ) );
   579→   579→        $wpdb->delete( $plans_table, array( 'id' => $id ), array( '%d' ) );
   580→   580→
   581→   581→        wp_safe_redirect( admin_url( 'admin.php?page=bv-plans&bv_deleted=1' ) );
   582→   582→        exit;
   583→   583→    }
