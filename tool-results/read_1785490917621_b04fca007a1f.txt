   152→    /**
   153→     * Verify admin nonce
   154→     */
   155→    private function verify_nonce() {
   156→        if ( ! check_ajax_referer( 'bv_admin_nonce', 'nonce', false ) ) {
   157→            wp_send_json_error( array( 'message' => 'Security check failed.' ), 403 );
   158→        }
   159→        if ( ! current_user_can( 'manage_options' ) ) {
   160→            wp_send_json_error( array( 'message' => 'Permission denied.' ), 403 );
   161→        }
   162→    }

   793→     * Get service data for editing
   794→     */
   795→    public function ajax_get_service() {
   796→        $this->verify_nonce();
   797→        global $wpdb;
   798→        $tables = $this->get_tables();
   799→
   800→        $id = intval( $_POST['id'] ?? 0 );
   801→        if ( $id <= 0 ) {
   802→            wp_send_json_error( array( 'message' => 'Invalid ID.' ) );
   803→        }
   804→
   805→        $service = $wpdb->get_row( $wpdb->prepare(
   806→            "SELECT * FROM {$tables['services']} WHERE id = %d",
   807→            $id
   808→        ), ARRAY_A );
   809→
   810→        if ( ! $service ) {
   811→            wp_send_json_error( array( 'message' => 'Service not found.' ) );
   812→        }
   813→
   814→        wp_send_json_success( $service );
   815→    }
