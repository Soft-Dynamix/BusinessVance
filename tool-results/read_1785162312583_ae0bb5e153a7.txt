   110→        global $post;
   111→        if ( is_a( $post, 'WP_Post' ) &&
   112→             ( has_shortcode( $post->post_content, 'businessvance_services' ) ||
   113→               has_shortcode( $post->post_content, 'businessvance_onceoff' ) ||
   114→               has_shortcode( $post->post_content, 'businessvance_subscriptions' ) )
   115→        ) {
   116→            wp_enqueue_style(
   117→                'bv-frontend-css',
   118→                BV_PLUGIN_URL . 'assets/css/frontend.css',
   119→                array(),
   120→                BV_VERSION
   121→            );
   122→        }
   123→    }

   343→    /**
   344→     * Get the correct URL for a service based on its type
   345→     */
   346→    private function get_service_url( $service ) {
   347→        if ( $service->woo_product_id ) {
   348→            return $this->get_woo_url( $service->woo_product_id );
   349→        }
   350→        if ( $service->service_type === 'quote' ) {
   351→            return '/contact';
   352→        }
   353→        if ( $service->service_type === 'booking' ) {
   354→            return '/booking';
   355→        }
   356→        return '#';
   357→    }
