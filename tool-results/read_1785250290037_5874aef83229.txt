    36→        /**
    37→         * Render the client portal shortcode output.
    38→         *
    39→         * @param array $atts Shortcode attributes.
    40→         * @return string HTML output.
    41→         */
    42→        public function render_portal( $atts ) {
    43→                if ( ! is_user_logged_in() ) {
    44→                        return $this->render_login_prompt();
    45→                }
    46→
    47→                global $wpdb;
    48→
    49→                $user_id    = get_current_user_id();
    50→                $user       = get_userdata( $user_id );
    51→                $first_name = get_user_meta( $user_id, 'first_name', true );
    52→                $last_name  = get_user_meta( $user_id, 'last_name', true );
    53→                $company    = get_user_meta( $user_id, 'billing_company', true );
    54→                $display_name = $user->display_name;
    55→
    56→                // Determine active project.
    57→                $active_project_id = isset( $_GET['project_id'] ) ? absint( $_GET['project_id'] ) : 0;
    58→                $active_tab        = isset( $_GET['tab'] ) ? sanitize_text_field( $_GET['tab'] ) : 'overview';
    59→
    60→                // Fetch all projects for this client.
    61→                $projects_table = $wpdb->prefix . 'bv_projects';
    62→                $projects = $wpdb->get_results(
    63→                        $wpdb->prepare(
    64→                                "SELECT * FROM {$projects_table} WHERE client_user_id = %d ORDER BY created_at DESC",
    65→                                $user_id
    66→                        )
    67→                );
    68→
    69→                if ( empty( $projects ) ) {
    70→                        return $this->render_no_projects( $display_name );
    71→                }
    72→
    73→                // If no active project, default to the first one.
    74→                if ( ! $active_project_id ) {
    75→                        $active_project_id = absint( $projects[0]->id );
    76→                }
    77→
    78→                // Verify the active project belongs to this user.
    79→                $active_project = null;
    80→                foreach ( $projects as $p ) {
    81→                        if ( absint( $p->id ) === $active_project_id ) {
    82→                                $active_project = $p;
    83→                                break;
    84→                        }
    85→                }
    86→                if ( ! $active_project ) {
    87→                        $active_project   = $projects[0];
    88→                        $active_project_id = absint( $active_project->id );
    89→                        $active_tab       = 'overview';
    90→                }
    91→
    92→                // Load project details.
    93→                $project_data = $this->load_project_data( $active_project_id );
    94→
    95→                // Valid tabs.
    96→                $valid_tabs = array( 'overview', 'agreement', 'questionnaire', 'documents', 'reports', 'messages' );
    97→                if ( ! in_array( $active_tab, $valid_tabs, true ) ) {
    98→                        $active_tab = 'overview';
    99→                }
   100→
   101→                // Build output.
   102→                ob_start();
   103→                ?>
   104→                <div class="bv-portal" id="bv-portal-root">
   105→                        <?php echo $this->render_portal_header( $display_name, $company ); ?>
   106→                        <?php echo $this->render_project_sidebar( $projects, $active_project_id ); ?>
   107→                        <?php echo $this->render_project_content( $active_project, $project_data, $active_tab ); ?>
   108→                </div>
   109→                <?php
   110→                return ob_get_clean();
   111→        }

   113→        /**
   114→         * Load all related data for a project.
   115→         *
   116→         * @param int $project_id The project ID.
   117→         * @return array Associative array of project data.
   118→         */
   119→        private function load_project_data( $project_id ) {
   120→                global $wpdb;
   121→
   122→                $data = array(
   123→                        'services'    => array(),
   124→                        'agreement'   => null,
   125→                        'documents'   => array(),
   126→                        'responses'   => array(),
   127→                        'reports'     => array(),
   128→                        'messages'    => array(),
   129→                        'notes_count' => 0,
   130→                );
   131→
   132→                // Services.
   133→                $data['services'] = $wpdb->get_results(
   134→                        $wpdb->prepare(
   135→                                "SELECT s.* FROM {$wpdb->prefix}bv_project_services ps
   136→                                  JOIN {$wpdb->prefix}bv_services s ON ps.service_id = s.id
   137→                                  WHERE ps.project_id = %d",
   138→                                $project_id
   139→                        )
   140→                );
   141→
   142→                // Agreement.
   143→                $agreements = $wpdb->get_results(
   144→                        $wpdb->prepare(
   145→                                "SELECT * FROM {$wpdb->prefix}bv_project_agreements
   146→                                  WHERE project_id = %d ORDER BY id DESC LIMIT 1",
   147→                                $project_id
   148→                        )
   149→                );
   150→                $data['agreement'] = ! empty( $agreements ) ? $agreements[0] : null;
   151→
   152→                // Documents.
   153→                $data['documents'] = $wpdb->get_results(
   154→                        $wpdb->prepare(
   155→                                "SELECT * FROM {$wpdb->prefix}bv_project_documents
   156→                                  WHERE project_id = %d ORDER BY created_at DESC",
   157→                                $project_id
   158→                        )
   159→                );
   160→
   161→                // Questionnaire responses.
   162→                $data['responses'] = $wpdb->get_results(
   163→                        $wpdb->prepare(
   164→                                "SELECT r.*, q.label, q.type FROM {$wpdb->prefix}bv_questionnaire_responses r
   165→                                  JOIN {$wpdb->prefix}bv_questionnaire_questions q ON r.question_id = q.id
   166→                                  WHERE r.project_id = %d",
   167→                                $project_id
   168→                        )
   169→                );
   170→
   171→                // Reports.
   172→                $data['reports'] = $wpdb->get_results(
   173→                        $wpdb->prepare(
   174→                                "SELECT * FROM {$wpdb->prefix}bv_project_reports
   175→                                  WHERE project_id = %d ORDER BY created_at DESC",
   176→                                $project_id
   177→                        )
   178→                );
   179→
   180→                // Messages.
   181→                $data['messages'] = $wpdb->get_results(
   182→                        $wpdb->prepare(
   183→                                "SELECT * FROM {$wpdb->prefix}bv_project_messages
   184→                                  WHERE project_id = %d ORDER BY created_at ASC",
   185→                                $project_id
   186→                        )
   187→                );
   188→
   189→                // Notes count (admin-only data, loaded but not displayed to client).
   190→                $data['notes_count'] = (int) $wpdb->get_var(
   191→                        $wpdb->prepare(
   192→                                "SELECT COUNT(*) FROM {$wpdb->prefix}bv_project_notes WHERE project_id = %d",
   193→                                $project_id
   194→                        )
   195→                );
   196→
   197→                return $data;
   198→        }

  1080→        /**
  1081→         * Verify the current user owns the given project.
  1082→         *
  1083→         * @param int $project_id Project ID to verify.
  1084→         * @return bool|WP_Error True if authorized, WP_Error on failure.
  1085→         */
  1086→        private function verify_project_ownership( $project_id ) {
  1087→                if ( ! is_user_logged_in() ) {
  1088→                        return new WP_Error( 'not_logged_in', __( 'You must be logged in.', 'businessvance' ) );
  1089→                }
  1090→
  1091→                global $wpdb;
  1092→                $project = $wpdb->get_row(
  1093→                        $wpdb->prepare(
  1094→                                "SELECT id, client_user_id FROM {$wpdb->prefix}bv_projects WHERE id = %d",
  1095→                                absint( $project_id )
  1096→                        )
  1097→                );
  1098→
  1099→                if ( ! $project ) {
  1100→                        return new WP_Error( 'project_not_found', __( 'Project not found.', 'businessvance' ) );
  1101→                }
  1102→
  1103→                if ( absint( $project->client_user_id ) !== get_current_user_id() ) {
  1104→                        return new WP_Error( 'not_authorized', __( 'You do not have access to this project.', 'businessvance' ) );
  1105→                }
  1106→
  1107→                return true;
  1108→        }

  1110→        /**
  1111→         * AJAX: Upload a document for a project.
  1112→         */
  1113→        public function ajax_upload_document() {
  1114→                check_ajax_referer( 'bv_portal_action', 'nonce' );
  1115→
  1116→                $project_id = isset( $_POST['project_id'] ) ? absint( $_POST['project_id'] ) : 0;
  1117→
  1118→                $ownership = $this->verify_project_ownership( $project_id );
  1119→                if ( is_wp_error( $ownership ) ) {
  1120→                        wp_send_json_error( array( 'message' => $ownership->get_error_message() ) );
  1121→                }
  1122→
  1123→                if ( empty( $_FILES['document'] ) ) {
  1124→                        wp_send_json_error( array( 'message' => __( 'No file uploaded.', 'businessvance' ) ) );
  1125→                }
  1126→
  1127→                $file     = $_FILES['document'];
  1128→                $allowed  = array( 'pdf', 'doc', 'docx', 'xls', 'xlsx', 'jpg', 'jpeg', 'png' );
  1129→                $file_ext = strtolower( pathinfo( $file['name'], PATHINFO_EXTENSION ) );
  1130→
  1131→                if ( ! in_array( $file_ext, $allowed, true ) ) {
  1132→                        wp_send_json_error( array( 'message' => __( 'File type not allowed.', 'businessvance' ) ) );
  1133→                }
  1134→
  1135→                // 10 MB max.
  1136→                if ( $file['size'] > 10 * 1024 * 1024 ) {
  1137→                        wp_send_json_error( array( 'message' => __( 'File exceeds the maximum size of 10MB.', 'businessvance' ) ) );
  1138→                }
  1139→
  1140→                $upload_dir = wp_upload_dir();
  1141→                $bv_dir     = $upload_dir['basedir'] . '/bv-documents';
  1142→
  1143→                if ( ! file_exists( $bv_dir ) ) {
  1144→                        wp_mkdir_p( $bv_dir );
  1145→                }
  1146→
  1147→                $unique_name = wp_unique_filename( $bv_dir, sanitize_file_name( $file['name'] ) );
  1148→                $dest_path   = $bv_dir . '/' . $unique_name;
  1149→
  1150→                if ( ! move_uploaded_file( $file['tmp_name'], $dest_path ) ) {
  1151→                        wp_send_json_error( array( 'message' => __( 'Failed to save the uploaded file.', 'businessvance' ) ) );
  1152→                }
  1153→
  1154→                $doc_name        = isset( $_POST['document_name'] ) ? sanitize_text_field( wp_unslash( $_POST['document_name'] ) ) : '';
  1155→                $doc_description = isset( $_POST['document_description'] ) ? sanitize_textarea_field( wp_unslash( $_POST['document_description'] ) ) : '';
  1156→
  1157→                if ( empty( $doc_name ) ) {
  1158→                        $doc_name = sanitize_file_name( $file['name'] );
  1159→                }
  1160→
  1161→                global $wpdb;
  1162→                $inserted = $wpdb->insert(
  1163→                        $wpdb->prefix . 'bv_project_documents',
  1164→                        array(
  1165→                                'project_id'  => $project_id,
  1166→                                'name'        => $doc_name,
  1167→                                'file_name'   => sanitize_file_name( $file['name'] ),
  1168→                                'file_path'   => 'bv-documents/' . $unique_name,
  1169→                                'file_size'   => $file['size'],
  1170→                                'description' => $doc_description,
  1171→                                'uploaded_by' => get_current_user_id(),
  1172→                                'created_at'  => current_time( 'mysql' ),
  1173→                        ),
  1174→                        array( '%d', '%s', '%s', '%s', '%d', '%s', '%d', '%s' )
  1175→                );
  1176→
  1177→                if ( ! $inserted ) {
  1178→                        // Clean up the file on DB failure.
  1179→                        @unlink( $dest_path );
  1180→                        wp_send_json_error( array( 'message' => __( 'Failed to save document record.', 'businessvance' ) ) );
  1181→                }
  1182→
  1183→                wp_send_json_success( array(
  1184→                        'message'  => __( 'Document uploaded successfully.', 'businessvance' ),
  1185→                        'document' => array(
  1186→                                'id'          => $wpdb->insert_id,
  1187→                                'name'        => $doc_name,
  1188→                                'file_name'   => sanitize_file_name( $file['name'] ),
  1189→                                'file_size'   => size_format( $file['size'] ),
  1190→                                'description' => $doc_description,
  1191→                                'created_at'  => current_time( 'mysql' ),
  1192→                        ),
  1193→                ) );
  1194→        }

  1402→        public function ajax_download_report() {
  1403→                check_ajax_referer( 'bv_portal_action', 'nonce' );
  1404→
  1405→                $project_id = isset( $_POST['project_id'] ) ? absint( $_POST['project_id'] ) : 0;
  1406→                $report_id  = isset( $_POST['report_id'] ) ? absint( $_POST['report_id'] ) : 0;
  1407→
  1408→                $ownership = $this->verify_project_ownership( $project_id );
  1409→                if ( is_wp_error( $ownership ) ) {
  1410→                        wp_send_json_error( array( 'message' => $ownership->get_error_message() ) );
  1411→                }
  1412→
  1413→                if ( empty( $report_id ) ) {
  1414→                        wp_send_json_error( array( 'message' => __( 'Invalid report.', 'businessvance' ) ) );
  1415→                }
  1416→
  1417→                global $wpdb;
  1418→
  1419→                $report = $wpdb->get_row(
  1420→                        $wpdb->prepare(
  1421→                                "SELECT * FROM {$wpdb->prefix}bv_project_reports WHERE id = %d AND project_id = %d",
  1422→                                $report_id,
  1423→                                $project_id
  1424→                        )
  1425→                );
  1426→
  1427→                if ( ! $report ) {
  1428→                        wp_send_json_error( array( 'message' => __( 'Report not found.', 'businessvance' ) ) );
  1429→                }
  1430→
  1431→                $file_path = ! empty( $report->file_path ) ? $report->file_path : '';
  1432→                $file_url  = ! empty( $report->file_url ) ? $report->file_url : '';
  1433→
  1434→                // Try to resolve the file location.
  1435→                $upload_dir = wp_upload_dir();
  1436→                $full_path  = '';
  1437→
  1438→                if ( ! empty( $file_path ) ) {
  1439→                        $relative = $file_path;
  1440→                        if ( 0 === strpos( $relative, 'bv-documents/' ) || 0 === strpos( $relative, '/bv-documents/' ) ) {
  1441→                                $full_path = $upload_dir['basedir'] . '/' . ltrim( $relative, '/' );
  1442→                        } elseif ( file_exists( $file_path ) ) {
  1443→                                $full_path = $file_path;
  1444→                        } else {
  1445→                                $full_path = $upload_dir['basedir'] . '/' . $relative;
  1446→                        }
  1447→                }
  1448→
  1449→                // If no file_path, try to use file_url.
  1450→                if ( empty( $full_path ) || ! file_exists( $full_path ) ) {
  1451→                        if ( ! empty( $file_url ) ) {
  1452→                                wp_send_json_success( array(
  1453→                                        'message'      => __( 'Report ready for download.', 'businessvance' ),
  1454→                                        'download_url' => $file_url,
  1455→                                        'redirect'     => true,
  1456→                                ) );
  1457→                        }
  1458→                        wp_send_json_error( array( 'message' => __( 'Report file not found. Please contact support.', 'businessvance' ) ) );
  1459→                }
  1460→
  1461→                // Serve the file directly.
  1462→                $file_name = ! empty( $report->title ) ? $report->title : ( ! empty( $report->name ) ? $report->name : 'report.pdf' );
  1463→
  1464→                // Clean file name for download.
  1465→                $file_name = sanitize_file_name( $file_name );
  1466→                $ext       = strtolower( pathinfo( $full_path, PATHINFO_EXTENSION ) );
  1467→                if ( $ext && false === strpos( $file_name, '.' . $ext ) ) {
  1468→                        $file_name .= '.' . $ext;
  1469→                }
  1470→
  1471→                // Determine MIME type.
  1472→                $mime_types = array(
  1473→                        'pdf'  => 'application/pdf',
  1474→                        'doc'  => 'application/msword',
  1475→                        'docx' => 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
  1476→                        'xls'  => 'application/vnd.ms-excel',
  1477→                        'xlsx' => 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
  1478→                        'png'  => 'image/png',
  1479→                        'jpg'  => 'image/jpeg',
  1480→                        'jpeg' => 'image/jpeg',
  1481→                );
  1482→                $mime = isset( $mime_types[ $ext ] ) ? $mime_types[ $ext ] : 'application/octet-stream';
  1483→
  1484→                wp_send_json_success( array(
  1485→                        'message'      => __( 'Report ready for download.', 'businessvance' ),
  1486→                        'download_url' => admin_url( 'admin-ajax.php?action=bv_portal_download_report&nonce=' . wp_create_nonce( 'bv_portal_action' ) . '&project_id=' . $project_id . '&report_id=' . $report_id . '&bv_serve=1' ),
  1487→                        'file_name'    => $file_name,
  1488→                ) );
  1489→        }
