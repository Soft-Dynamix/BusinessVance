   142→   142→function SortableRow({ id, children }: { id: string; children: React.ReactNode }) {
   143→   143→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id });

   149→   149→    zIndex: isDragging ? 50 : undefined,
   150→   150→  };
   151→   151→  return (
   152→   152→    <tr ref={setNodeRef} style={style} {...attributes}>
   153→   153→      {React.Children.map(children, (child, i) => {
   154→   154→        if (i === 0) return <td className="text-center p-2"><button className="cursor-grab active:cursor-grabbing text-[#9ca3af] hover:text-[#50575e] transition-colors" {...listeners} aria-label="Drag to reorder"><GripVertical size={18} /></button></td>;
   155→   155→        return child;
   156→   156→      })}
   157→   157→    </tr>
   158→   158→  );
