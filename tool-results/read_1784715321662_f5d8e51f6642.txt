   142→function SortableRow({ id, children }: { id: string; children: React.ReactNode }) {
   143→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id });
   144→  const style: React.CSSProperties = {
   145→    transform: CSS.Transform.toString(transform),
   146→    transition,
   147→    opacity: isDragging ? 0.4 : 1,
   148→    background: isDragging ? '#FEF3C7' : undefined,
   149→    zIndex: isDragging ? 50 : undefined,
   150→  };
   151→  return (
   152→    <tr ref={setNodeRef} style={style} {...attributes}>
   153→      {React.Children.map(children, (child, i) => {
   154→        if (i === 0) return <td className="text-center p-2"><button className="cursor-grab active:cursor-grabbing text-[#9ca3af] hover:text-[#50575e] transition-colors" {...listeners} aria-label="Drag to reorder"><GripVertical size={18} /></button></td>;
   155→        return child;
   156→      })}
   157→    </tr>
   158→  );
   159→}

   358→  const handleServiceReorder = async (event: DragEndEvent) => {
   359→    const { active, over } = event;

   362→    const oldIdx = ids.indexOf(String(active.id));
   363→    const newIdx = ids.indexOf(String(over.id));

   365→    const reordered = arrayMove(services, oldIdx, newIdx);
   366→    setServices(reordered);
   367→    try {
   368→      await fetch('/api/services/reorder', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ items: reordered.map((s, i) => ({ id: s.id, displayOrder: i })) }) });
   369→    } catch { showToast('Reorder failed'); }
   370→  };
