   152→    /**
   153→     * Verify admin nonce
   154→     */
   155→    private function verify_nonce() {
   156→        if ( ! check_ajax_referer( 'bv_admin_nonce', 'nonce', false ) ) {
   157→            wp_send_json_error( array( 'message' => 'Security check failed.' ), 403 );
   158→        }
   159→        if ( ! current_user_can( 'manage_options' ) ) {
   160→            wp_send_json_error( array( 'message' => 'Permission denied.' ), 403 );
   161→        }
   162→    }
