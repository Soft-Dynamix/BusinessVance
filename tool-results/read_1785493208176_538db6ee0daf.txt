   119→    public function ajax_get_templates() {
   120→        $this->verify_nonce();
   121→
   122→        global $wpdb;
   123→        $t = $this->get_tables();
   124→
   125→        $results = $wpdb->get_results(
   126→            "SELECT t.*, 
   127→                (SELECT COUNT(*) FROM {$t['sections']} WHERE template_id = t.id) AS section_count,
   128→                (SELECT COUNT(*) FROM {$t['questions']} qq 
   129→                    JOIN {$t['sections']} s ON qq.section_id = s.id 
   130→                    WHERE s.template_id = t.id) AS question_count 
   131→            FROM {$t['templates']} t 
   132→            ORDER BY t.created_at DESC"
   133→        );
   134→
   135→        wp_send_json_success( array( 'templates' => $results ) );
   136→    }

   138→    /* ==========================================================================
   139→     * AJAX: Get single template with all sections and questions
   140→     * ========================================================================== */
   141→
   142→    public function ajax_get_template() {
   143→        $this->verify_nonce();
   144→
   145→        global $wpdb;
   146→        $t = $this->get_tables();
   147→
   148→        $template_id = absint( $_POST['template_id'] );
   149→        if ( ! $template_id ) {
   150→            wp_send_json_error( array( 'message' => 'Invalid template ID.' ) );
   151→        }
   152→
   153→        $template = $wpdb->get_row( $wpdb->prepare(
   154→            "SELECT * FROM {$t['templates']} WHERE id = %d",
   155→            $template_id
   156→        ) );
   157→
   158→        if ( ! $template ) {
   159→            wp_send_json_error( array( 'message' => 'Template not found.' ) );
   160→        }
   161→
   162→        $sections = $wpdb->get_results( $wpdb->prepare(
   163→            "SELECT * FROM {$t['sections']} WHERE template_id = %d ORDER BY display_order ASC, id ASC",
   164→            $template_id
   165→        ) );
   166→
   167→        if ( $sections ) {
   168→            foreach ( $sections as $section ) {
   169→                $questions = $wpdb->get_results( $wpdb->prepare(
   170→                    "SELECT * FROM {$t['questions']} WHERE section_id = %d ORDER BY display_order ASC, id ASC",
   171→                    $section->id
   172→                ) );
   173→
   174→                if ( $questions ) {
   175→                    foreach ( $questions as $question ) {
   176→                        $question->options = json_decode( $question->options, true );
   177→                        if ( ! is_array( $question->options ) ) {
   178→                            $question->options = array();
   179→                        }
   180→                    }
   181→                } else {
   182→                    $questions = array();
   183→                }
   184→
   185→                $section->questions = $questions;
   186→            }
   187→        } else {
   188→            $sections = array();
   189→        }
   190→
   191→        $template->sections = $sections;
   192→
   193→        wp_send_json_success( array( 'template' => $template ) );
   194→    }

   197→     * AJAX: Save (create or update) template

   200→    public function ajax_save_template() {
   201→        $this->verify_nonce();
   202→
   203→        global $wpdb;
   204→        $t = $this->get_tables();
   205→
   206→        $id          = isset( $_POST['id'] ) ? absint( $_POST['id'] ) : 0;
   207→        $name        = isset( $_POST['name'] ) ? sanitize_text_field( wp_unslash( $_POST['name'] ) ) : '';
   208→        $slug        = isset( $_POST['slug'] ) ? sanitize_title( wp_unslash( $_POST['slug'] ) ) : sanitize_title( $name );
   209→        $description = isset( $_POST['description'] ) ? sanitize_textarea_field( wp_unslash( $_POST['description'] ) ) : '';
   210→        $status      = isset( $_POST['status'] ) ? sanitize_text_field( $_POST['status'] ) : 'draft';
   211→
   212→        if ( empty( $name ) ) {
   213→            wp_send_json_error( array( 'message' => 'Template name is required.' ) );
   214→        }
   215→
   216→        if ( ! in_array( $status, array( 'draft', 'published' ), true ) ) {
   217→            $status = 'draft';
   218→        }
   219→
   220→        // Check for duplicate slug
   221→        if ( $id ) {
   222→            $existing = $wpdb->get_var( $wpdb->prepare(
   223→                "SELECT COUNT(*) FROM {$t['templates']} WHERE slug = %s AND id != %d",
   224→                $slug,
   225→                $id
   226→            ) );
   227→        } else {
   228→            $existing = $wpdb->get_var( $wpdb->prepare(
   229→                "SELECT COUNT(*) FROM {$t['templates']} WHERE slug = %s",
   230→                $slug
   231→            ) );
   232→        }
   233→
   234→        if ( (int) $existing > 0 ) {
   235→            wp_send_json_error( array( 'message' => 'A template with this slug already exists. Please use a different name.' ) );
   236→        }
   237→
   238→        $data = array(
   239→            'name'        => $name,
   240→            'slug'        => $slug,
   241→            'description' => $description,
   242→            'status'      => $status,
   243→            'updated_at'  => current_time( 'mysql' ),
   244→        );
   245→        $format = array( '%s', '%s', '%s', '%s', '%s' );
   246→
   247→        if ( $id ) {
   248→            $wpdb->update( $t['templates'], $data, array( 'id' => $id ), $format, array( '%d' ) );
   249→        } else {
   250→            $data['created_at'] = current_time( 'mysql' );
   251→            $format[] = '%s';
   252→            $wpdb->insert( $t['templates'], $data, $format );
   253→            $id = $wpdb->insert_id;
   254→        }
   255→
   256→        if ( $wpdb->last_error ) {
   257→            wp_send_json_error( array( 'message' => 'Database error: ' . $wpdb->last_error ) );
   258→        }
   259→
   260→        wp_send_json_success( array(
   261→            'template_id' => $id,
   262→            'message'     => 'Template saved successfully.',
   263→        ) );
   264→    }

   267→     * AJAX: Delete template (cascade to sections and questions)
   268→     * ========================================================================== */
   269→
   270→    public function ajax_delete_template() {
   271→        $this->verify_nonce();
   272→
   273→        global $wpdb;
   274→        $t = $this->get_tables();
   275→
   276→        $id = isset( $_POST['id'] ) ? absint( $_POST['id'] ) : 0;
   277→        if ( ! $id ) {
   278→            wp_send_json_error( array( 'message' => 'Invalid template ID.' ) );
   279→        }
   280→
   281→        // Get section IDs for this template
   282→        $section_ids = $wpdb->get_col( $wpdb->prepare(
   283→            "SELECT id FROM {$t['sections']} WHERE template_id = %d",
   284→            $id
   285→        ) );
   286→
   287→        // Delete all questions in those sections
   288→        if ( ! empty( $section_ids ) ) {
   289→            $section_id_list = implode( ',', array_map( 'absint', $section_ids ) );
   290→            $wpdb->query( "DELETE FROM {$t['questions']} WHERE section_id IN ({$section_id_list})" );
   291→        }
   292→
   293→        // Delete all sections
   294→        $wpdb->delete( $t['sections'], array( 'template_id' => $id ), array( '%d' ) );
   295→
   296→        // Delete template
   297→        $wpdb->delete( $t['templates'], array( 'id' => $id ), array( '%d' ) );
   298→
   299→        wp_send_json_success( array( 'message' => 'Template deleted successfully.' ) );
   300→    }

   387→    /* ==========================================================================
   388→     * AJAX: Save (create or update) question
   389→     * ========================================================================== */
   390→
   391→    public function ajax_save_question() {
   392→        $this->verify_nonce();
   393→
   394→        global $wpdb;
   395→        $t = $this->get_tables();
   396→
   397→        $id           = isset( $_POST['id'] ) ? absint( $_POST['id'] ) : 0;
   398→        $section_id   = isset( $_POST['section_id'] ) ? absint( $_POST['section_id'] ) : 0;
   399→        $type         = isset( $_POST['type'] ) ? sanitize_text_field( $_POST['type'] ) : 'text';
   400→        $label        = isset( $_POST['label'] ) ? sanitize_text_field( wp_unslash( $_POST['label'] ) ) : '';
   401→        $placeholder  = isset( $_POST['placeholder'] ) ? sanitize_text_field( wp_unslash( $_POST['placeholder'] ) ) : '';
   402→        $is_required  = isset( $_POST['is_required'] ) ? (int) $_POST['is_required'] : 0;
   403→        $help_text    = isset( $_POST['help_text'] ) ? sanitize_text_field( wp_unslash( $_POST['help_text'] ) ) : '';
   404→        $options_text = isset( $_POST['options_text'] ) ? wp_unslash( $_POST['options_text'] ) : '';
   405→
   406→        $allowed_types = array(
   407→            'text', 'textarea', 'number', 'email', 'phone', 'date',
   408→            'select', 'radio', 'checkbox', 'heading', 'paragraph', 'file',
   409→        );
   410→
   411→        if ( ! in_array( $type, $allowed_types, true ) ) {
   412→            $type = 'text';
   413→        }
   414→
   415→        if ( ! $section_id ) {
   416→            wp_send_json_error( array( 'message' => 'Section ID is required.' ) );
   417→        }
   418→
   419→        if ( empty( $label ) ) {
   420→            wp_send_json_error( array( 'message' => 'Question label is required.' ) );
   421→        }
   422→
   423→        // Convert options_text to JSON array
   424→        $options = array();
   425→        if ( in_array( $type, array( 'select', 'radio', 'checkbox' ), true ) && ! empty( $options_text ) ) {
   426→            $lines = explode( "\n", $options_text );
   427→            foreach ( $lines as $line ) {
   428→                $line = trim( $line );
   429→                if ( '' === $line ) {
   430→                    continue;
   431→                }
   432→                if ( strpos( $line, '|' ) !== false ) {
   433→                    $parts = explode( '|', $line, 2 );
   434→                    $options[] = array(
   435→                        'value' => sanitize_text_field( trim( $parts[0] ) ),
   436→                        'label' => sanitize_text_field( trim( $parts[1] ) ),
   437→                    );
   438→                } else {
   439→                    $options[] = array(
   440→                        'value' => sanitize_title( $line ),
   441→                        'label' => sanitize_text_field( $line ),
   442→                    );
   443→                }
   444→            }
   445→        }
   446→
   447→        $options_json = wp_json_encode( $options );
   448→
   449→        $data = array(
   450→            'section_id'  => $section_id,
   451→            'type'        => $type,
   452→            'label'       => $label,
   453→            'placeholder' => $placeholder,
   454→            'is_required' => $is_required,
   455→            'options'     => $options_json,
   456→            'help_text'   => $help_text,
   457→        );
   458→        $format = array( '%d', '%s', '%s', '%s', '%d', '%s', '%s' );
   459→
   460→        if ( $id ) {
   461→            $wpdb->update( $t['questions'], $data, array( 'id' => $id ), $format, array( '%d' ) );
   462→        } else {
   463→            // Get the next display_order
   464→            $max_order = (int) $wpdb->get_var( $wpdb->prepare(
   465→                "SELECT COALESCE(MAX(display_order), -1) FROM {$t['questions']} WHERE section_id = %d",
   466→                $section_id
   467→            ) );
   468→            $data['display_order'] = $max_order + 1;
   469→            $format[] = '%d';
   470→            $data['created_at'] = current_time( 'mysql' );
   471→            $format[] = '%s';
   472→            $wpdb->insert( $t['questions'], $data, $format );
   473→            $id = $wpdb->insert_id;
   474→        }
   475→
   476→        if ( $wpdb->last_error ) {
   477→            wp_send_json_error( array( 'message' => 'Database error: ' . $wpdb->last_error ) );
   478→        }
   479→
   480→        $question = $wpdb->get_row( $wpdb->prepare(
   481→            "SELECT * FROM {$t['questions']} WHERE id = %d",
   482→            $id
   483→        ) );
   484→
   485→        if ( $question ) {
   486→            $question->options = json_decode( $question->options, true );
   487→            if ( ! is_array( $question->options ) ) {
   488→                $question->options = array();
   489→            }
   490→        }
   491→
   492→        wp_send_json_success( array(
   493→            'question' => $question,
   494→            'message'  => 'Question saved successfully.',
   495→        ) );
   496→    }

   518→    /* ==========================================================================
   519→     * AJAX: Reorder sections or questions
   520→     * ========================================================================== */
   521→
   522→    public function ajax_reorder() {
   523→        $this->verify_nonce();
   524→
   525→        global $wpdb;
   526→        $t = $this->get_tables();
   527→
   528→        $type = isset( $_POST['type'] ) ? sanitize_text_field( $_POST['type'] ) : '';
   529→        $ids  = isset( $_POST['ids'] ) ? sanitize_text_field( $_POST['ids'] ) : '';
   530→
   531→        if ( ! in_array( $type, array( 'section', 'question' ), true ) ) {
   532→            wp_send_json_error( array( 'message' => 'Invalid reorder type.' ) );
   533→        }
   534→
   535→        $id_array = array_map( 'absint', explode( ',', $ids ) );
   536→        $id_array = array_filter( $id_array );
   537→
   538→        if ( empty( $id_array ) ) {
   539→            wp_send_json_error( array( 'message' => 'No IDs provided.' ) );
   540→        }
   541→
   542→        if ( 'section' === $type ) {
   543→            $table = $t['sections'];
   544→        } else {
   545→            $table = $t['questions'];
   546→        }
   547→
   548→        foreach ( $id_array as $order => $item_id ) {
   549→            $wpdb->update(
   550→                $table,
   551→                array( 'display_order' => $order ),
   552→                array( 'id' => $item_id ),
   553→                array( '%d' ),
   554→                array( '%d' )
   555→            );
   556→        }
   557→
   558→        wp_send_json_success( array( 'message' => 'Order updated successfully.' ) );
   559→    }
